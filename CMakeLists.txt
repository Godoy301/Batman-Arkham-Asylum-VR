cmake_minimum_required(VERSION 3.21)
project(BatmanAsylumVR VERSION 0.1.0 LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Asylum requires MSVC Win32: -G Visual Studio 17 2022 -A Win32")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX)
add_subdirectory(third_party)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(batman_asylum SHARED ${CORE_SOURCES} src/hash.cpp src/game/adapter_registry.cpp
 src/game/asylum/adapter.cpp src/game/asylum/d3d9_bridge.cpp src/game/asylum/render_anchors.cpp)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/bvr_version.h"
 "#pragma once\n#define BVR_VERSION \"${PROJECT_VERSION}\"\n#define BVR_BUILD_ID \"Asylum-development-A0a\"\n")
target_include_directories(batman_asylum PRIVATE src "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(batman_asylum PRIVATE minhook imgui openxr_loader d3d11 d3d12 dxgi d3dcompiler dbghelp windowscodecs ole32 bcrypt)
target_compile_definitions(batman_asylum PRIVATE BVR_WITH_OPENXR)
add_library(d3d9 SHARED src/d3d9_proxy.cpp src/d3d9_proxy.def)
add_library(xinput1_3 SHARED src/proxy.cpp src/proxy.def)
add_executable(aa_xr_config32 tools/xr_config32.cpp)
target_link_libraries(aa_xr_config32 PRIVATE openxr_loader)
add_executable(asylum_offline_check tests/offline_check.cpp)
enable_testing()
add_test(NAME foreign_host_and_proxy_guard COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>")
add_test(NAME d3d9_gpu_transport COMMAND asylum_offline_check "$<TARGET_FILE:batman_asylum>" "$<TARGET_FILE:d3d9>" "$<TARGET_FILE:xinput1_3>" gpu)
add_test(NAME graphics_profile_rollback COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/graphics_profile_check.ps1")
